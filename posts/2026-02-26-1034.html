Fourier Analysis: Decomposing Functions into Waves

Motivation: In 1807, Joseph Fourier submitted a revolutionary paper to the Paris Institute, claiming that any function could be expressed as a sum of sine and cosine waves. This insight, initially rejected by Lagrange and Laplace, would transform mathematics, physics, and engineering. Fourier analysis provides the mathematical foundation for signal processing, quantum mechanics, and the study of partial differential equations.

Definitions

  • Fourier Series: For a periodic function \\(f\\) with period \\(2\\pi\\), the Fourier series is \\(f(x) \\sim \\sum_{n=-\\infty}^{\\infty} c_n e^{inx}\\) where \\(c_n = \\frac{1}{2\\pi} \\int_{-\\pi}^{\\pi} f(x) e^{-inx} dx\\).
  • Fourier Transform: For \\(f \\in L^1(\\mathbb{R})\\), the Fourier transform is \\(\\hat{f}(\\xi) = \\int_{-\\infty}^{\\infty} f(x) e^{-2\\pi i x \\xi} dx\\).
  • Plancherel's Theorem: The Fourier transform extends to a unitary operator on \\(L^2(\\mathbb{R})\\), preserving the inner product.
  • Convolution: \\((f * g)(x) = \\int_{-\\infty}^{\\infty} f(y)g(x-y)dy\\), with \\(\\widehat{f * g} = \\hat{f} \\cdot \\hat{g}\\).

Theorem: Fourier Inversion

Theorem: If \\(f \\in L^1(\\mathbb{R})\\) and \\(\\hat{f} \\in L^1(\\mathbb{R})\\), then for almost every \\(x\\): \\[f(x) = \\int_{-\\infty}^{\\infty} \\hat{f}(\\xi) e^{2\\pi i x \\xi} d\\xi\\]

Proof Sketch: The proof uses an approximate identity approach. Consider \\(f * K_\\epsilon\\) where \\(K_\\epsilon\\) is a Gaussian with variance \\(\\epsilon\\). The Fourier transform of a Gaussian is a Gaussian, and \\(\\widehat{f * K_\\epsilon} = \\hat{f} \\cdot \\widehat{K_\\epsilon}\\). As \\(\\epsilon \\to 0\\), \\(f * K_\\epsilon \\to f\\) in \\(L^1\\), while \\(\\widehat{K_\\epsilon}\\) approaches 1. The dominated convergence theorem allows interchange of limit and integral, yielding the inversion formula.

Notation Guide

  • \\(\\hat{f}\\) or \\(\\mathcal{F}(f)\\): Fourier transform
  • \\(c_n\\): Fourier coefficients
  • \\(\\check{f}\\): Inverse Fourier transform
  • \\(\\mathcal{S}(\\mathbb{R})\\): Schwartz space of rapidly decreasing functions
  • \\(\\mathcal{F}^{-1}\\): Inverse Fourier transform operator

import numpy as np
import matplotlib.pyplot as plt
from numpy.fft import fft, ifft, fftfreq

class FourierAnalyzer:
    """
    Tools for Fourier analysis and signal decomposition.
    """
    def __init__(self, signal, dt=0.01):
        """
        Initialize with signal and time step.
        
        Args:
            signal: 1D array of signal values
            dt: time step between samples
        """
        self.signal = np.array(signal)
        self.dt = dt
        self.N = len(signal)
        self.t = np.arange(self.N) * dt
    
    def compute_fourier_series(self, n_harmonics=10):
        """
        Compute Fourier series coefficients for periodic signal.
        
        Returns:
            a0: DC component
            an: cosine coefficients
            bn: sine coefficients
        """
        T = self.N * self.dt
        omega = 2 * np.pi / T
        
        # DC component
        a0 = (2 / self.N) * np.sum(self.signal)
        
        an = []
        bn = []
        
        for n in range(1, n_harmonics + 1):
            cos_terms = self.signal * np.cos(n * omega * self.t)
            sin_terms = self.signal * np.sin(n * omega * self.t)
            
            an.append((2 / self.N) * np.sum(cos_terms))
            bn.append((2 / self.N) * np.sum(sin_terms))
        
        return a0, np.array(an), np.array(bn)
    
    def reconstruct_signal(self, a0, an, bn, n_terms=None):
        """
        Reconstruct signal from Fourier coefficients.
        """
        if n_terms is None:
            n_terms = len(an)
        
        T = self.N * self.dt
        omega = 2 * np.pi / T
        
        reconstruction = np.ones_like(self.t) * a0 / 2
        
        for n in range(1, min(n_terms + 1, len(an) + 1)):
            reconstruction += an[n-1] * np.cos(n * omega * self.t)
            reconstruction += bn[n-1] * np.sin(n * omega * self.t)
        
        return reconstruction
    
    def compute_fft(self):
        """
        Compute Fast Fourier Transform.
        """
        fft_vals = fft(self.signal)
        freqs = fftfreq(self.N, self.dt)
        
        # Return only positive frequencies
        pos_mask = freqs >= 0
        return freqs[pos_mask], np.abs(fft_vals[pos_mask])
    
    def plot_analysis(self, n_harmonics=10):
        """
        Visualize Fourier analysis of signal.
        """
        fig, axes = plt.subplots(3, 1, figsize=(12, 10))
        
        # Original signal
        axes[0].plot(self.t, self.signal, 'b-', linewidth=1.5, label='Original')
        
        # Fourier series reconstruction
        a0, an, bn = self.compute_fourier_series(n_harmonics)
        reconstruction = self.reconstruct_signal(a0, an, bn, n_harmonics)
        axes[0].plot(self.t, reconstruction, 'r--', linewidth=1.5, 
                     label=f'Fourier Series ({n_harmonics} terms)')
        axes[0].set_xlabel('Time')
        axes[0].set_ylabel('Amplitude')
        axes[0].set_title('Signal and Fourier Series Reconstruction')
        axes[0].legend()
        axes[0].grid(True, alpha=0.3)
        
        # Fourier coefficients
        n_vals = np.arange(1, len(an) + 1)
        axes[1].stem(n_vals[:n_harmonics], an[:n_harmonics], 
                     linefmt='b-', markerfmt='bo', basefmt=' ', label='Cosine (an)')
        axes[1].stem(n_vals[:n_harmonics], bn[:n_harmonics], 
                     linefmt='r-', markerfmt='ro', basefmt=' ', label='Sine (bn)')
        axes[1].set_xlabel('Harmonic Number n')
        axes[1].set_ylabel('Coefficient Value')
        axes[1].set_title('Fourier Series Coefficients')
        axes[1].legend()
        axes[1].grid(True, alpha=0.3)
        
        # FFT spectrum
        freqs, magnitude = self.compute_fft()
        axes[2].plot(freqs[:self.N//2], magnitude[:self.N//2], 'g-', linewidth=1.5)
        axes[2].set_xlabel('Frequency (Hz)')
        axes[2].set_ylabel('Magnitude')
        axes[2].set_title('FFT Frequency Spectrum')
        axes[2].grid(True, alpha=0.3)
        
        plt.tight_layout()
        return fig

Differential Forms: The Language of Modern Geometry

Motivation: Differential forms provide a coordinate-independent framework for multivariable calculus, generalizing the notions of line integrals, surface integrals, and volume integrals. Élie Cartan developed this theory in the early 20th century, unifying vector calculus, tensor analysis, and algebraic topology. Forms are essential in modern physics, particularly in general relativity and gauge theory.

Definitions

  • Differential k-form: A smooth section of the \\(k\\)-th exterior power of the cotangent bundle \\(\\Lambda^k T^*M\\), written locally as \\(\\omega = \\sum_{|I|=k} f_I dx^I\\).
  • Exterior Derivative: \\(d: \\Omega^k(M) \\to \\Omega^{k+1}(M)\\) satisfying \\(d^2 = 0\\), \\(d(\\alpha \\wedge \\beta) = d\\alpha \\wedge \\beta + (-1)^{|\\alpha|} \\alpha \\wedge d\\beta\\).
  • Pullback: For \\(f: M \\to N\\), the pullback \\(f^*: \\Omega^k(N) \\to \\Omega^k(M)\\) is defined by \\((f^*\\omega)_p(v_1, ..., v_k) = \\omega_{f(p)}(df_p(v_1), ..., df_p(v_k))\\).
  • Closed Form: \\(\\omega\\) is closed if \\(d\\omega = 0\\).
  • Exact Form: \\(\\omega\\) is exact if \\(\\omega = d\\eta\\) for some \\(\\eta\\).

Theorem: Stokes' Theorem

Theorem: Let \\(M\\) be an oriented \\(n\\)-dimensional manifold with boundary \\(\\partial M\\), and let \\(\\omega\\) be a compactly supported \\((n-1)\\)-form on \\(M\\). Then: \\[\\int_M d\\omega = \\int_{\\partial M} \\omega\\]

Proof Sketch: The proof proceeds by partition of unity, reducing to the case where \\(\\omega\\) is supported in a coordinate chart. In local coordinates, \\(d\\omega\\) involves partial derivatives, and the fundamental theorem of calculus applies. The boundary orientation convention ensures the signs work out correctly. The key insight is that interior contributions cancel due to opposite orientations on shared boundaries.

Notation Guide

  • \\(\\Omega^k(M)\\): Space of \\(k\\)-forms on \\(M\\)
  • \\(dx^I = dx^{i_1} \\wedge \\cdots \\wedge dx^{i_k}\\): Basic \\(k\\)-form
  • \\(\\wedge\\): Wedge product (antisymmetric tensor product)
  • \\(H^k_{dR}(M)\\): \\(k\\)-th de Rham cohomology group
  • \\(\\iota_X\\): Interior product (contraction with vector field \\(X\\))

import numpy as np
from typing import Callable, Tuple, List
import sympy as sp

class DifferentialForm:
    """
    Represents a differential form in R^n using symbolic computation.
    """
    def __init__(self, degree: int, components: dict, coords: List[sp.Symbol]):
        """
        Initialize a differential form.
        
        Args:
            degree: Degree of the form (k)
            components: Dict mapping multi-indices to coefficient functions
            coords: List of coordinate symbols
        """
        self.degree = degree
        self.components = components
        self.coords = coords
        self.n = len(coords)
    
    def exterior_derivative(self):
        """
        Compute the exterior derivative d(omega).
        """
        if self.degree >= self.n:
            return DifferentialForm(self.degree + 1, {}, self.coords)
        
        new_components = {}
        
        for idx, coeff in self.components.items():
            # d(f dx^I) = sum_j (df/dx_j) dx^j ^ dx^I
            for j in range(self.n):
                partial = sp.diff(coeff, self.coords[j])
                if partial != 0:
                    new_idx = tuple(sorted([j] + list(idx)))
                    
                    # Compute sign from antisymmetry
                    sign = 1
                    for i in idx:
                        if i < j:
                            sign *= -1
                    
                    new_coeff = sign * partial
                    
                    if new_idx in new_components:
                        new_components[new_idx] += new_coeff
                    else:
                        new_components[new_idx] = new_coeff
        
        return DifferentialForm(self.degree + 1, new_components, self.coords)
    
    def wedge_product(self, other):
        """
        Compute wedge product self ^ other.
        """
        new_degree = self.degree + other.degree
        new_components = {}
        
        for idx1, coeff1 in self.components.items():
            for idx2, coeff2 in other.components.items():
                # Check for repeated indices
                if set(idx1) & set(idx2):
                    continue
                
                new_idx = tuple(sorted(list(idx1) + list(idx2)))
                
                # Compute sign from antisymmetry
                sign = 1
                for i in idx1:
                    for j in idx2:
                        if i > j:
                            sign *= -1
                
                new_coeff = sign * coeff1 * coeff2
                
                if new_idx in new_components:
                    new_components[new_idx] += new_coeff
                else:
                    new_components[new_idx] = new_coeff
        
        return DifferentialForm(new_degree, new_components, self.coords)
    
    def __str__(self):
        terms = []
        for idx, coeff in sorted(self.components.items()):
            dx_str = " ".join([f"dx{i}" for i in idx])
            if coeff == 1:
                terms.append(dx_str)
            elif coeff == -1:
                terms.append(f"- {dx_str}")
            else:
                terms.append(f"{coeff} {dx_str}")
        return " + ".join(terms) if terms else "0"
    
    def evaluate(self, point: dict) -> float:
        """
        Evaluate form at a point (returns components dict).
        """
        result = {}
        for idx, coeff in self.components.items():
            result[idx] = float(coeff.subs(point))
        return result

Algebraic Combinatorics: The RSK Correspondence

Motivation: The Robinson-Schensted-Knuth (RSK) correspondence, discovered in the 1930s-70s, provides a fundamental bijection between matrices with non-negative integer entries and pairs of semistandard Young tableaux of the same shape. This correspondence connects representation theory, symmetric functions, and the theory of algorithms, with applications ranging from random matrix theory to the analysis of sorting algorithms.

Definitions

  • Young Diagram: A collection of boxes arranged in left-justified rows with weakly decreasing row lengths \\(\\lambda = (\\lambda_1 \\geq \\lambda_2 \\geq \\cdots)\\).
  • Semistandard Young Tableau (SSYT): A filling of a Young diagram with positive integers that are weakly increasing along rows and strictly increasing down columns.
  • Schensted Insertion: An algorithm that inserts a number into a tableau by bumping larger entries to the next row.
  • Knuth Equivalence: Two words are Knuth equivalent if they have the same insertion tableau under RSK.

Theorem: RSK Correspondence

Theorem: There is a bijection between \\(m \\times n\\) matrices \\(A = (a_{ij})\\) with non-negative integer entries and pairs \\((P, Q)\\) of semistandard Young tableaux of the same shape, where \\(P\\) has entries in \\([n]\\) and \\(Q\\) has entries in \\([m]\\). If \\(A\\) has sum of entries equal to \\(k\\), then \\(P\\) and \\(Q\\) each have \\(k\\) boxes.

Proof Sketch: The construction proceeds by row insertion. For each entry \\(a_{ij}\\), insert \\(j\\) into \\(P\\) using Schensted insertion \\(a_{ij}\\) times, recording each insertion in \\(Q\\) by placing \\(i\\) in the newly created box. The insertion algorithm maintains the tableau properties: weakly increasing rows and strictly increasing columns. The inverse map reconstructs the matrix by reverse insertion, using \\(Q\\) to determine which row of \\(A\\) contributed each box.

Notation Guide

  • \\(\\lambda \\vdash n\\): Partition of \\(n\\)
  • \\(\\text{SSYT}(\\lambda)\\): Set of semistandard Young tableaux of shape \\(\\lambda\\)
  • \\(\\text{sh}(T)\\): Shape of tableau \\(T\\)
  • \\(\\text{ins}(w)\\): Insertion tableau of word \\(w\\)
  • \\(\\text{rec}(w)\\): Recording tableau of word \\(w\\)

Python Implementation

class YoungTableau:
    """
    Represents a Young tableau with insertion operations.
    """
    def __init__(self, shape=None, entries=None):
        """
        Initialize tableau.
        
        Args:
            shape: List of row lengths
            entries: 2D list of entries (optional)
        """
        if shape is None:
            shape = []
        self.shape = list(shape)
        
        if entries is None:
            self.rows = [[] for _ in range(len(shape))]
        else:
            self.rows = [list(row) for row in entries]
    
    def insert(self, x):
        """
        Schensted insert x into the tableau.
        Returns the position where a new box was created (if any).
        """
        row_idx = 0
        current = x
        bumped = True
        new_box_pos = None
        
        while bumped:
            if row_idx >= len(self.rows):
                # Create new row
                self.rows.append([current])
                self.shape.append(1)
                new_box_pos = (row_idx, 0)
                bumped = False
            else:
                row = self.rows[row_idx]
                # Find position to insert (weakly increasing)
                pos = 0
                while pos < len(row) and row[pos] <= current:
                    pos += 1
                
                if pos == len(row):
                    # Append to end
                    row.append(current)
                    if row_idx < len(self.shape):
                        self.shape[row_idx] += 1
                    else:
                        self.shape.append(1)
                    new_box_pos = (row_idx, pos)
                    bumped = False
                else:
                    # Bump the larger element
                    bumped_val = row[pos]
                    row[pos] = current
                    current = bumped_val
                    row_idx += 1
        
        return new_box_pos
    
    def is_valid_ssyt(self):
        """
        Check if tableau is a valid semistandard Young tableau.
        """
        # Check rows weakly increasing
        for row in self.rows:
            for i in range(len(row) - 1):
                if row[i] > row[i+1]:
                    return False
        
        # Check columns strictly increasing
        max_len = max(len(row) for row in self.rows) if self.rows else 0
        for col in range(max_len):
            prev = -1
            for row in self.rows:
                if col < len(row):
                    if row[col] <= prev and prev != -1:
                        return False
                    prev = row[col]
        
        return True
    
    def __str__(self):
        lines = []
        for row in self.rows:
            lines.append("  ".join(str(x) for x in row))
        return "\\n".join(lines)
    
    def copy(self):
        return YoungTableau(self.shape[:], [row[:] for row in self.rows])


def rsk_correspondence(matrix):
    """
    Compute RSK correspondence for a matrix.
    
    Args:
        matrix: 2D list of non-negative integers
    
    Returns:
        (P, Q): Pair of Young tableaux
    """
    m = len(matrix)
    n = len(matrix[0]) if m > 0 else 0
    
    P = YoungTableau()
    Q = YoungTableau()
    
    for i in range(m):
        for j in range(n):
            count = matrix[i][j]
            for _ in range(count):
                # Insert j+1 into P
                new_pos = P.insert(j + 1)
                
                # Insert i+1 into Q at the same position
                if new_pos:
                    r, c = new_pos
                    while r >= len(Q.rows):
                        Q.rows.append([])
                        Q.shape.append(0)
                    while c >= len(Q.rows[r]):
                        Q.rows[r].append(None)
                    Q.rows[r][c] = i + 1
                    Q.shape[r] = max(Q.shape[r], c + 1)
    
    return P, Q

Markov Processes: Memoryless Stochastic Evolution

Motivation: Markov processes model systems that evolve randomly without memory of their past, given their present state. Named after Andrey Markov, who introduced them in 1906 to study sequences of dependent random variables, these processes appear everywhere from physics (Brownian motion) to economics (option pricing) to biology (population genetics). The Markov property simplifies analysis while capturing essential dynamics.

Definitions

  • Markov Property: \\(P(X_{t+1} = j | X_t = i, X_{t-1}, ..., X_0) = P(X_{t+1} = j | X_t = i)\\).
  • Transition Matrix: \\(P = (p_{ij})\\) where \\(p_{ij} = P(X_{t+1} = j | X_t = i)\\), satisfying \\(\\sum_j p_{ij} = 1\\).
  • Stationary Distribution: A probability vector \\(\\pi\\) satisfying \\(\\pi P = \\pi\\).
  • Irreducibility: A chain is irreducible if every state is reachable from every other state.
  • Aperiodicity: A state has period \\(d\\) if returns are possible only at multiples of \\(d\\); aperiodic if \\(d = 1\\).

Theorem: Convergence to Stationary Distribution

Theorem: Let \\(P\\) be the transition matrix of a finite-state Markov chain that is irreducible and aperiodic. Then there exists a unique stationary distribution \\(\\pi\\), and for any initial distribution \\(\\mu\\): \\[\\lim_{n \\to \\infty} \\mu P^n = \\pi\\]

Proof Sketch: Irreducibility ensures the existence of a unique stationary distribution via the Perron-Frobenius theorem. Aperiodicity guarantees that \\(P^n\\) has all positive entries for sufficiently large \\(n\\). The convergence follows from the coupling method: run two chains with different initial distributions until they meet. The meeting time is almost surely finite, and once coupled, the chains remain identical. This shows the total variation distance between distributions decreases to zero.

Notation Guide

  • \\(X_t\\): State at time \\(t\\)
  • \\(P^n_{ij}\\): \\(n\\)-step transition probability
  • \\(\\pi_i\\): Stationary probability of state \\(i\\)
  • \\(T_i\\): Return time to state \\(i\\)
  • \\(\\mathbb{E}_i[T_i] = 1/\\pi_i\\): Expected return time

import numpy as np
import matplotlib.pyplot as plt
from typing import Tuple, List

class MarkovChain:
    """
    Finite-state Markov chain with analysis tools.
    """
    def __init__(self, transition_matrix: np.ndarray):
        """
        Initialize Markov chain.
        
        Args:
            transition_matrix: n x n stochastic matrix
        """
        self.P = np.array(transition_matrix, dtype=float)
        self.n_states = self.P.shape[0]
        
        # Verify stochastic
        assert np.allclose(self.P.sum(axis=1), 1), "Rows must sum to 1"
        assert np.all(self.P >= 0), "Probabilities must be non-negative"
    
    def is_irreducible(self) -> bool:
        """
        Check if chain is irreducible (all states communicate).
        """
        # Compute (I + P)^(n-1) and check all entries positive
        Q = np.eye(self.n_states)
        for _ in range(self.n_states - 1):
            Q = Q @ (np.eye(self.n_states) + self.P)
        return np.all(Q > 0)
    
    def is_aperiodic(self) -> bool:
        """
        Check if chain is aperiodic.
        """
        # Chain is aperiodic if some power of P has all positive entries
        P_power = self.P.copy()
        for _ in range(self.n_states ** 2):
            if np.all(P_power > 0):
                return True
            P_power = P_power @ self.P
        return False
    
    def stationary_distribution(self) -> np.ndarray:
        """
        Compute stationary distribution using eigenvalue method.
        """
        # Find left eigenvector with eigenvalue 1
        eigenvalues, eigenvectors = np.linalg.eig(self.P.T)
        
        # Find eigenvalue closest to 1
        idx = np.argmin(np.abs(eigenvalues - 1))
        stationary = np.real(eigenvectors[:, idx])
        
        # Normalize to probability distribution
        stationary = stationary / stationary.sum()
        return stationary
    
    def simulate(self, initial_state: int, n_steps: int) -> List[int]:
        """
        Simulate trajectory of the Markov chain.
        """
        trajectory = [initial_state]
        current = initial_state
        
        for _ in range(n_steps):
            current = np.random.choice(self.n_states, p=self.P[current])
            trajectory.append(current)
        
        return trajectory
    
    def n_step_distribution(self, initial_dist: np.ndarray, n: int) -> np.ndarray:
        """
        Compute distribution after n steps.
        """
        return initial_dist @ np.linalg.matrix_power(self.P, n)
    
    def mixing_time(self, epsilon: float = 0.01, max_steps: int = 10000) -> int:
        """
        Estimate mixing time: steps needed for TV distance < epsilon.
        """
        pi = self.stationary_distribution()
        
        # Start from worst case (point mass)
        for i in range(self.n_states):
            dist = np.eye(self.n_states)[i]
            for t in range(max_steps):
                dist = dist @ self.P
                tv_dist = 0.5 * np.sum(np.abs(dist - pi))
                if tv_dist < epsilon:
                    return t
        
        return max_steps
    
    def plot_convergence(self, initial_dists: List[np.ndarray], max_steps: int = 100):
        """
        Visualize convergence to stationary distribution.
        """
        pi = self.stationary_distribution()
        
        fig, axes = plt.subplots(2, 1, figsize=(12, 10))
        
        # Plot total variation distance over time
        for i, init in enumerate(initial_dists):
            tv_distances = []
            dist = init.copy()
            for _ in range(max_steps):
                tv_dist = 0.5 * np.sum(np.abs(dist - pi))
                tv_distances.append(tv_dist)
                dist = dist @ self.P
            
            axes[0].plot(tv_distances, label=f'Initial {i+1}')
        
        axes[0].axhline(y=0, color='k', linestyle='--', alpha=0.3)
        axes[0].set_xlabel('Time Steps')

        axes[0].set_title('Convergence to Stationary Distribution')

        axes[0].grid(True, alpha=0.3)

        axes[1].set_ylabel('Stationary Probability')
        axes[1].set_title('Stationary Distribution')
        axes[1].grid(True, alpha=0.3)
        
        plt.tight_layout()
        return fig

Statistical Learning Theory: PAC Learning

Motivation: Probably Approximately Correct (PAC) learning, introduced by Leslie Valiant in 1984, provides a computational framework for understanding when machines can learn from data. PAC theory establishes rigorous bounds on the amount of data needed for a learning algorithm to generalize well, connecting computational complexity with statistical inference. It forms the theoretical foundation of modern machine learning.

Definitions

  • Concept Class \\(\\mathcal{C}\\): A set of functions \\(c: \\mathcal{X} \\to \\{0, 1\\}\\) (binary classifiers).
  • Hypothesis Class \\(\\mathcal{H}\\): The set of functions the learning algorithm can output.
  • Generalization Error: \\(R(h) = P(h(X) \\neq c(X))\\) for target concept \\(c\\).
  • Empirical Risk: \\(\\hat{R}_n(h) = \\frac{1}{n} \\sum_{i=1}^n \\mathbf{1}_{h(x_i) \\neq c(x_i)}\\).
  • VC Dimension: The largest \\(d\\) such that \\(\\mathcal{H}\\) shatters some set of \\(d\\) points.

Theorem: PAC Learning Bound

Theorem: Let \\(\\mathcal{H}\\) be a hypothesis class with finite VC dimension \\(d\\). For any \\(\\epsilon, \\delta > 0\\), if the sample size satisfies: \\[n \\geq \\frac{C}{\\epsilon^2} \\left(d + \\log\\frac{1}{\\delta}\\right)\\] then with probability at least \\(1 - \\delta\\), any empirical risk minimizer \\(\\hat{h}\\) satisfies \\(R(\\hat{h}) \\leq R^* + \\epsilon\\), where \\(R^* = \\inf_{h \\in \\mathcal{H}} R(h)\\).

Proof Sketch: The proof uses uniform convergence arguments. By the VC inequality, the empirical risk converges uniformly to the true risk over the hypothesis class. The key is bounding the covering number of \\(\\mathcal{H}\\) using the growth function, which is polynomial in \\(n\\) when VC dimension is finite. Sauer's lemma shows the growth function is bounded by \\(O(n^d)\\), leading to the sample complexity bound via concentration inequalities.

Notation Guide

  • \\(\\mathcal{X}\\): Input space (feature space)
  • \\(\\mathcal{Y}\\): Output space (labels)
  • \\(\\mathcal{D}\\): Distribution over \\(\\mathcal{X} \\times \\mathcal{Y}\\)
  • \\(S = \\{(x_i, y_i)\\}_{i=1}^n\\): Training sample
  • \\(\\Pi_{\\mathcal{H}}(n)\\): Growth function (shatter function)

import numpy as np

from typing import Callable, List, Tuple

class PACLearning:
    """
    Demonstrates PAC learning concepts and bounds.
    """
    def __init__(self, true_concept: Callable, input_dim: int):
        """
        Initialize PAC learning framework.
        
        Args:
            true_concept: Function mapping x -> {0, 1}
            input_dim: Dimension of input space
        """
        self.true_concept = true_concept
        self.d = input_dim
    
    def generate_sample(self, n: int) -> Tuple[np.ndarray, np.ndarray]:
        """
        Generate n labeled examples.
        """
        X = np.random.randn(n, self.d)
        y = np.array([self.true_concept(x) for x in X])
        return X, y
    
    def empirical_risk(self, h: Callable, X: np.ndarray, y: np.ndarray) -> float:
        """
        Compute empirical risk of hypothesis h.
        """
        predictions = np.array([h(x) for x in X])
        return np.mean(predictions != y)
    
    def true_risk(self, h: Callable, n_test: int = 10000) -> float:
        """
        Estimate true risk by Monte Carlo.
        """
        X_test, y_test = self.generate_sample(n_test)
        return self.empirical_risk(h, X_test, y_test)
    
    def vc_bound_sample_size(self, epsilon: float, delta: float, 
                              vc_dim: int) -> int:
        """
        Compute sample size bound from VC theory.
        
        n >= (C/epsilon^2) * (d + log(1/delta))
        """
        C = 1.0  # Constant factor
        return int(np.ceil((C / epsilon**2) * (vc_dim + np.log(1/delta))))
    
    def uniform_convergence_bound(self, n: int, delta: float, 
                                   vc_dim: int) -> float:
        """
        Bound on sup_h |R(h) - R_n(h)| with probability 1-delta.
        """
        # Simplified bound: O(sqrt(d log(n/d) / n))
        if n <= vc_dim:
            return 1.0
        return np.sqrt(vc_dim * np.log(n / vc_dim) / n) + np.sqrt(np.log(1/delta) / (2*n))
    
    def learning_curve(self, sample_sizes: List[int], 
                       hypothesis_class: str = 'linear',

            'test_errors': [],
            'std_train': [],
            'std_test': []
        }

            train_errs = []
            test_errs = []
            
            for _ in range(n_trials):
                X_train, y_train = self.generate_sample(n)
                X_test, y_test = self.generate_sample(1000)
                
                # Fit linear classifier

                model.fit(X_train, y_train)
                
                h = lambda x: model.predict(x.reshape(1, -1))[0]
                
                train_errs.append(self.empirical_risk(h, X_train, y_train))
                test_errs.append(self.true_risk(h))
            
            results['train_errors'].append(np.mean(train_errs))
            results['test_errors'].append(np.mean(test_errs))
            results['std_train'].append(np.std(train_errs))
            results['std_test'].append(np.std(test_errs))
        
        return results
    
    def plot_learning_curve(self, results: dict, vc_dim: int = None):

                    yerr=results['std_test'], label='Test Error (True Risk)',
                    marker='s', capsize=3)
        
        # Plot theoretical bound

            bounds = [self.uniform_convergence_bound(n, delta, vc_dim)

        ax.set_xlabel('Sample Size n')
        ax.set_ylabel('Error')
        ax.set_title('Learning Curve: Empirical vs True Risk')
        ax.legend()
        ax.grid(True, alpha=0.3)
        ax.set_xscale('log')
        
        return fig

Numerical PDEs: Finite Element Methods

Motivation: The finite element method (FEM) revolutionized the numerical solution of partial differential equations by providing a flexible framework for handling complex geometries and varying material properties. Developed by engineers in the 1950s-60s and placed on rigorous mathematical foundations by mathematicians like Babuška and Brezzi, FEM is now the dominant approach for structural analysis, fluid dynamics, and electromagnetics.

Definitions

  • Weak Formulation: Find \\(u \\in V\\) such that \\(a(u, v) = F(v)\\) for all \\(v \\in V\\), where \\(a(\\cdot, \\cdot)\\) is a bilinear form and \\(F\\) is a linear functional.
  • Sobolev Space \\(H^1(\\Omega)\\): Functions with square-integrable first derivatives.
  • Finite Element Space \\(V_h\\): Piecewise polynomial functions on a mesh.
  • Galerkin Approximation: Find \\(u_h \\in V_h\\) such that \\(a(u_h, v_h) = F(v_h)\\) for all \\(v_h \\in V_h\\).

Theorem: Céa's Lemma

Theorem: Let \\(a(\\cdot, \\cdot)\\) be a continuous, coercive bilinear form on Hilbert space \\(V\\), and let \\(V_h \\subset V\\) be a finite-dimensional subspace. If \\(u\\) solves the weak problem and \\(u_h\\) is the Galerkin approximation, then: \\[\\|u - u_h\\|_V \\leq \\frac{M}{\\alpha} \\inf_{v_h \\in V_h} \\|u - v_h\\|_V\\] where \\(M\\) is the continuity constant and \\(\\alpha\\) is the coercivity constant.

Proof Sketch: The Galerkin orthogonality property \\(a(u - u_h, v_h) = 0\\) for all \\(v_h \\in V_h\\) is key. For any \\(v_h \\in V_h\\), write \\(u - u_h = (u - v_h) + (v_h - u_h)\\). Apply coercivity to the left side and continuity to the right, using orthogonality to eliminate the cross term. The result follows by taking the infimum over \\(v_h\\).

Notation Guide

  • \\(\\Omega\\): Domain of the PDE
  • \\(\\partial \\Omega\\): Boundary of domain
  • \\(\\mathcal{T}_h\\): Triangulation/mesh
  • \\(h\\): Mesh size parameter
  • \\(P_k\\): Space of polynomials of degree \\(\\leq k\\)

import numpy as np
import matplotlib.pyplot as plt
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import spsolve
from typing import Callable, Tuple, List

class FiniteElement1D:
    """
    1D Finite Element solver for -u'' = f with Dirichlet BCs.
    """
    def __init__(self, domain: Tuple[float, float], n_elements: int):
        """
        Initialize FEM on 1D domain.
        
        Args:
            domain: (a, b) interval endpoints
            n_elements: Number of elements
        """
        self.a, self.b = domain
        self.n_el = n_elements
        self.n_nodes = n_elements + 1
        self.h = (self.b - self.a) / n_elements
        
        # Node coordinates
        self.nodes = np.linspace(self.a, self.b, self.n_nodes)
    
    def basis_function(self, i: int, x: float) -> float:
        """
        Evaluate hat function phi_i at point x.
        """
        if i < 0 or i >= self.n_nodes:
            return 0.0
        
        xi = self.nodes[i]
        
        if x < xi - self.h or x > xi + self.h:
            return 0.0
        elif x < xi:
            return (x - (xi - self.h)) / self.h
        else:
            return ((xi + self.h) - x) / self.h
    
    def basis_derivative(self, i: int, x: float) -> float:
        """
        Evaluate derivative of hat function phi_i at point x.
        """
        if i < 0 or i >= self.n_nodes:
            return 0.0
        
        xi = self.nodes[i]
        
        if x < xi - self.h or x > xi + self.h:
            return 0.0
        elif x < xi:
            return 1.0 / self.h
        else:
            return -1.0 / self.h
    
    def assemble_stiffness_matrix(self) -> np.ndarray:
        """
        Assemble stiffness matrix A where A_ij = integral(phi_i' * phi_j').
        """
        # For linear elements on uniform mesh:
        # A is tridiagonal with pattern: [-1, 2, -1] / h
        
        A = np.zeros((self.n_nodes, self.n_nodes))
        
        for i in range(self.n_nodes):
            if i > 0:
                A[i, i-1] = -1.0 / self.h
            A[i, i] = 2.0 / self.h
            if i < self.n_nodes - 1:
                A[i, i+1] = -1.0 / self.h
        
        return A
    
    def assemble_load_vector(self, f: Callable) -> np.ndarray:
        """
        Assemble load vector F where F_i = integral(f * phi_i).
        """
        F = np.zeros(self.n_nodes)
        
        # Use trapezoidal rule for integration
        for i in range(self.n_nodes):
            # Integrate over support of phi_i
            left = max(0, i-1)
            right = min(self.n_nodes-1, i+1)
            
            for e in range(left, right):
                x_mid = (self.nodes[e] + self.nodes[e+1]) / 2
                F[i] += f(x_mid) * self.basis_function(i, x_mid) * self.h
        
        return F
    
    def solve(self, f: Callable, bc_left: float, bc_right: float) -> np.ndarray:
        """
        Solve -u'' = f with u(a) = bc_left, u(b) = bc_right.
        """
        A = self.assemble_stiffness_matrix()
        F = self.assemble_load_vector(f)
        
        # Apply Dirichlet boundary conditions
        # Modify first and last rows
        A[0, :] = 0
        A[0, 0] = 1
        F[0] = bc_left
        
        A[-1, :] = 0
        A[-1, -1] = 1
        F[-1] = bc_right
        
        # Solve linear system
        u = np.linalg.solve(A, F)
        
        return u
    
    def evaluate_solution(self, u: np.ndarray, x: float) -> float:
        """
        Evaluate FEM solution at arbitrary point x.
        """
        result = 0.0
        for i in range(self.n_nodes):
            result += u[i] * self.basis_function(i, x)
        return result
    
    def compute_error(self, u: np.ndarray, u_exact: Callable, 
                      n_quad: int = 100) -> float:
        """
        Compute L2 error between FEM solution and exact solution.
        """
        x_quad = np.linspace(self.a, self.b, n_quad)
        errors = []
        
        for x in x_quad:
            u_fem = self.evaluate_solution(u, x)
            u_ex = u_exact(x)
            errors.append((u_fem - u_ex)**2)
        
        return np.sqrt(np.mean(errors) * (self.b - self.a))
    
    def plot_solution(self, u: np.ndarray, u_exact: Callable = None):
        """
        Visualize FEM solution.
        """
        fig, ax = plt.subplots(figsize=(10, 6))
        
        # Plot FEM solution
        x_fine = np.linspace(self.a, self.b, 200)
        u_fine = [self.evaluate_solution(u, x) for x in x_fine]
        
        ax.plot(x_fine, u_fine, 'b-', linewidth=2, label='FEM Solution')
        ax.scatter(self.nodes, u, color='red', s=50, zorder=5, 
                   label='Nodal Values')
        
        # Plot exact solution if provided
        if u_exact:
            u_exact_fine = [u_exact(x) for x in x_fine]
            ax.plot(x_fine, u_exact_fine, 'g--', linewidth=1.5, 
                    label='Exact Solution')
        
        ax.set_xlabel('x')
        ax.set_ylabel('u(x)')
        ax.set_title(f'Finite Element Solution (n={self.n_el} elements)')
        ax.legend()
        ax.grid(True, alpha=0.3)
        
        return fig

Knot Polynomials: The HOMFLY Polynomial

Historical Context: The discovery of the Jones polynomial in 1984 sparked a revolution in knot theory. Vaughan Jones's work, for which he received the Fields Medal in 1990, was followed by the discovery of the HOMFLY polynomial (named after its six independent discoverers: Hoste, Ocneanu, Millett, Freyd, Lickorish, and Yetter, with Przytycki and Traczyk also contributing). This two-variable polynomial generalizes both the Alexander and Jones polynomials.

Definitions

  • Oriented Link Diagram: A projection of a link with over/under crossings and orientation arrows on each component.
  • Writhe: For an oriented diagram \\(D\\), \\(w(D) = \\sum_{\\text{crossings}} \\epsilon(c)\\) where \\(\\epsilon = \\pm 1\\) is the crossing sign.
  • Reidemeister Moves: Three local moves that generate ambient isotopy of knots.
  • Skein Relation: A linear relation between polynomials of links differing at a single crossing.

Theorem: Existence and Uniqueness of HOMFLY

Theorem: There exists a unique invariant \\(P(L)\\) of oriented links, taking values in \\(\\mathbb{Z}[a^{\\pm 1}, z^{\\pm 1}]\\), satisfying:
(1) Normalization: \\(P(\\text{unknot}) = 1\\)
(2) Skein relation: \\(a^{-1} P(L_+) - a P(L_-) = z P(L_0)\\)
where \\(L_+, L_-, L_0\\) are links identical except at one crossing.

Proof Sketch: Existence is proved by defining \\(P\\) recursively on crossing number using the skein relation. Any crossing can be "resolved" to reduce complexity. Uniqueness follows from the fact that any two sequences of resolutions yield the same result, which is verified by checking consistency on the Reidemeister moves. The polynomial is invariant under all three Reidemeister moves, hence an ambient isotopy invariant.

Notation Guide

  • \\(L_+\\): Positive crossing
  • \\(L_-\\): Negative crossing
  • \\(L_0\\): Smoothed crossing (oriented)
  • \\(\\nabla_K(t)\\): Alexander polynomial (specialization)
  • \\(V_K(t)\\): Jones polynomial (specialization)

from typing import Dict, Tuple
import sympy as sp

class KnotDiagram:
    """
    Represents an oriented knot diagram for polynomial computation.
    """
    def __init__(self, crossings: list, orientation: list):
        """
        Initialize knot diagram.
        
        Args:
            crossings: List of crossing information
            orientation: Orientation of each strand
        """
        self.crossings = crossings
        self.orientation = orientation
        self.n_crossings = len(crossings)
    
    def writhe(self) -> int:
        """
        Compute writhe of the diagram.
        """
        return sum(c['sign'] for c in self.crossings)
    
    def is_trivial(self) -> bool:
        """
        Check if diagram represents unknot.
        """
        return self.n_crossings == 0

        """

        if key in self.cache:
            return self.cache[key]
        
        # Base case: unknot
        if diagram.is_trivial():
            return sp.Integer(1)
        
        # Apply skein relation at first crossing
        if diagram.n_crossings > 0:
            crossing = diagram.crossings[0]
            sign = crossing['sign']
            
            # Create L_+, L_-, L_0 diagrams
            # This is simplified - full implementation needs proper diagram manipulation

            if diagram.n_crossings == 1:
                # Single crossing diagrams

Modular Arithmetic: The Chinese Remainder Theorem

Historical Context: The Chinese Remainder Theorem (CRT) first appeared in the 3rd-century mathematical text "Sunzi Suanjing" by the Chinese mathematician Sunzi. The problem asks: "There are certain things whose number is unknown. If we count them by threes, there is a remainder 2; by fives, there is a remainder 3; and by sevens, there is a remainder 2. What will be the number of things?" The general solution was later developed by Qin Jiushao in 1247.

Definitions

  • Congruence: \\(a \\equiv b \\pmod{n}\\) means \\(n | (a - b)\\).
  • Ring of Integers Modulo n: \\(\\mathbb{Z}/n\\mathbb{Z} = \\{0, 1, ..., n-1\\}\\) with modular arithmetic.
  • Coprime: \\(\\gcd(m, n) = 1\\).
  • Euler's Totient Function: \\(\\varphi(n) = |\\{k : 1 \\leq k \\leq n, \\gcd(k, n) = 1\\}|\\).

Theorem: Chinese Remainder Theorem

Theorem: Let \\(n_1, n_2, ..., n_k\\) be pairwise coprime positive integers, and let \\(N = n_1 n_2 \\cdots n_k\\). For any integers \\(a_1, a_2, ..., a_k\\), there exists a unique integer \\(x\\) modulo \\(N\\) satisfying: \\[x \\equiv a_i \\pmod{n_i} \\text{ for all } i = 1, ..., k\\] Moreover, the solution is given by: \\[x = \\sum_{i=1}^k a_i N_i M_i \\pmod{N}\\] where \\(N_i = N/n_i\\) and \\(M_i = N_i^{-1} \\pmod{n_i}\\).

Proof Sketch: Existence: For each \\(i\\), since \\(\\gcd(N_i, n_i) = 1\\), the inverse \\(M_i\\) exists by Bézout's identity. The term \\(a_i N_i M_i\\) is congruent to \\(a_i\\) modulo \\(n_i\\) and \\(0\\) modulo \\(n_j\\) for \\(j \\neq i\\). The sum satisfies all congruences. Uniqueness: If \\(x\\) and \\(y\\) are both solutions, then \\(n_i | (x - y)\\) for all \\(i\\), so \\(N | (x - y)\\) since the \\(n_i\\) are coprime.

Notation Guide

  • \\(\\mathbb{Z}/n\\mathbb{Z}\\): Integers modulo \\(n\\)
  • \\((\\mathbb{Z}/n\\mathbb{Z})^*\\): Multiplicative group of units
  • \\(a^{-1} \\pmod{n}\\): Modular multiplicative inverse
  • \\(\\cong\\): Isomorphism of rings/groups

from typing import List, Tuple
import math

class ChineseRemainderTheorem:
    """
    Implementation of Chinese Remainder Theorem and related algorithms.
    """
    
    @staticmethod
    def extended_gcd(a: int, b: int) -> Tuple[int, int, int]:
        """
        Extended Euclidean algorithm.
        Returns (g, x, y) where g = gcd(a, b) = ax + by.
        """
        if b == 0:
            return (a, 1, 0)
        
        g, x1, y1 = ChineseRemainderTheorem.extended_gcd(b, a % b)
        x = y1
        y = x1 - (a // b) * y1
        
        return (g, x, y)
    
    @staticmethod
    def mod_inverse(a: int, n: int) -> int:
        """
        Compute modular multiplicative inverse of a modulo n.
        """
        g, x, _ = ChineseRemainderTheorem.extended_gcd(a % n, n)
        
        if g != 1:
            raise ValueError(f"Inverse does not exist: gcd({a}, {n}) = {g}")
        
        return (x % n + n) % n
    
    @staticmethod
    def solve(moduli: List[int], remainders: List[int]) -> int:
        """
        Solve system of congruences using CRT.
        
        Args:
            moduli: List of pairwise coprime moduli n_i
            remainders: List of remainders a_i
        
        Returns:
            Solution x modulo N = product of moduli
        """
        if len(moduli) != len(remainders):
            raise ValueError("Number of moduli and remainders must match")
        
        # Verify pairwise coprime
        for i in range(len(moduli)):
            for j in range(i + 1, len(moduli)):
                if math.gcd(moduli[i], moduli[j]) != 1:
                    raise ValueError(f"Moduli {moduli[i]} and {moduli[j]} are not coprime")
        
        N = math.prod(moduli)
        result = 0
        
        for a_i, n_i in zip(remainders, moduli):
            N_i = N // n_i
            M_i = ChineseRemainderTheorem.mod_inverse(N_i, n_i)
            result += a_i * N_i * M_i
        
        return result % N
    
    @staticmethod
    def euler_totient(n: int) -> int:
        """
        Compute Euler's totient function phi(n).
        """
        result = n
        p = 2
        
        while p * p <= n:
            if n % p == 0:
                while n % p == 0:
                    n //= p
                result -= result // p
            p += 1
        
        if n > 1:
            result -= result // n
        
        return result
    
    @staticmethod
    def solve_iterative(moduli: List[int], remainders: List[int]) -> int:
        """
        Solve CRT iteratively (for large number of moduli).
        """
        x = remainders[0]
        m = moduli[0]
        
        for i in range(1, len(moduli)):
            # Solve: x + m*k ≡ remainders[i] (mod moduli[i])
            # => m*k ≡ remainders[i] - x (mod moduli[i])
            
            rhs = (remainders[i] - x) % moduli[i]
            m_inv = ChineseRemainderTheorem.mod_inverse(m % moduli[i], moduli[i])
            k = (rhs * m_inv) % moduli[i]
            
            x = x + m * k
            m = m * moduli[i]
            x = x % m
        
        return x

Theorem: Newton's Method Convergence

Theorem: Let \\(f\\) be twice continuously differentiable with \\(f(r) = 0\\) and \\(f'(r) \\neq 0\\). If \\(x_0\\) is sufficiently close to \\(r\\), then Newton's iteration: \\[x_{n+1} = x_n - \\frac{f(x_n)}{f'(x_n)}\\] converges quadratically to \\(r\\), i.e., \\(|x_{n+1} - r| \\leq C|x_n - r|^2\\) for some constant \\(C\\).

Proof Sketch: By Taylor's theorem, \\(f(r) = f(x_n) + f'(x_n)(r - x_n) + \\frac{1}{2}f''(\\xi_n)(r - x_n)^2\\). Since \\(f(r) = 0\\), rearranging gives \\(r - x_{n+1} = -\\frac{f''(\\xi_n)}{2f'(x_n)}(r - x_n)^2\\). Near \\(r\\), the coefficient is bounded, yielding quadratic convergence.

import numpy as np
import matplotlib.pyplot as plt
from typing import Callable, List

class NewtonMethods:
    """
    Implementation of Newton's mathematical methods.
    """
    
    @staticmethod
    def newton_method(f: Callable, df: Callable, x0: float, 
                      tol: float = 1e-10, max_iter: int = 100) -> dict:
        """
        Newton's method for finding roots.
        
        Args:
            f: Function to find root of
            df: Derivative of f
            x0: Initial guess
            tol: Tolerance for convergence
            max_iter: Maximum iterations
        
        Returns:
            Dictionary with root, iterations, and history
        """
        x = x0
        history = [x]
        
        for i in range(max_iter):
            fx = f(x)
            
            if abs(fx) < tol:
                return {
                    'root': x,
                    'iterations': i,
                    'converged': True,
                    'history': history
                }
            
            dfx = df(x)
            
            if abs(dfx) < 1e-15:
                return {
                    'root': x,
                    'iterations': i,
                    'converged': False,
                    'history': history,
                    'error': 'Derivative too small'
                }
            
            x = x - fx / dfx
            history.append(x)
        
        return {
            'root': x,
            'iterations': max_iter,
            'converged': False,
            'history': history
        }
    
    @staticmethod
    def generalized_binomial(alpha: float, n: int) -> List[float]:
        """
        Compute coefficients of (1 + x)^alpha using generalized binomial theorem.
        
        (1 + x)^alpha = sum_{k=0}^infty C(alpha, k) x^k
        where C(alpha, k) = alpha(alpha-1)...(alpha-k+1) / k!
        """
        coeffs = []
        coeff = 1.0
        
        for k in range(n + 1):
            coeffs.append(coeff)
            # Update for next coefficient
            coeff *= (alpha - k) / (k + 1)
        
        return coeffs
    
    @staticmethod
    def newton_divided_difference(x: List[float], y: List[float]) -> np.ndarray:
        """
        Compute Newton's divided difference coefficients for interpolation.
        """
        n = len(x)
        coef = np.array(y, dtype=float)
        
        for j in range(1, n):
            for i in range(n - 1, j - 1, -1):
                coef[i] = (coef[i] - coef[i - 1]) / (x[i] - x[i - j])
        
        return coef
    
    @staticmethod
    def newton_interpolate(x_data: List[float], coef: np.ndarray, x: float) -> float:
        """
        Evaluate Newton interpolating polynomial at x.
        """
        n = len(coef)
        result = coef[-1]
        
        for i in range(n - 2, -1, -1):
            result = result * (x - x_data[i]) + coef[i]
        
        return result
    
    @staticmethod
    def gravitational_force(m1: float, m2: float, r: float, G: float = 6.674e-11) -> float:
        """
        Newton's law of universal gravitation.
        """
        return G * m1 * m2 / (r ** 2)
    
    @staticmethod
    def planetary_orbit(a: float, e: float, n_points: int = 1000) -> Tuple[np.ndarray, np.ndarray]:
        """
        Generate points on elliptical orbit (Newton's solution to Kepler's problem).
        
        Args:
            a: Semi-major axis
            e: Eccentricity
            n_points: Number of points
        
        Returns:
            x, y coordinates of orbit
        """
        theta = np.linspace(0, 2 * np.pi, n_points)
        r = a * (1 - e**2) / (1 + e * np.cos(theta))
        
        x = r * np.cos(theta)
        y = r * np.sin(theta)
        
        return x, y

The Golden Spiral: Nature's Mathematical Pattern

Historical Context: The golden ratio \\(\\phi = \\frac{1 + \\sqrt{5}}{2} \\approx 1.618\\) has fascinated mathematicians since antiquity. Euclid called it the "extreme and mean ratio." The golden spiral, a logarithmic spiral with growth factor \\(\\phi\\) per quarter turn, appears throughout nature—in the arrangement of seeds in sunflowers, the shape of nautilus shells, and the structure of galaxies. While often exaggerated in popular culture, the underlying mathematics reveals deep connections between geometry and growth.

Definitions

  • Golden Ratio: \\(\\phi = \\frac{1 + \\sqrt{5}}{2}\\) satisfies \\(\\phi^2 = \\phi + 1\\).
  • Golden Rectangle: A rectangle with aspect ratio \\(\\phi\\); removing a square leaves another golden rectangle.
  • Logarithmic Spiral: \\(r = ae^{b\\theta}\\) where \\(a, b\\) are constants.
  • Golden Angle: \\(\\theta_g = \\frac{2\\pi}{\\phi^2} = 2\\pi(1 - \\frac{1}{\\phi}) \\approx 137.5°\\).

Theorem: Self-Similarity of the Golden Spiral

Theorem: The logarithmic spiral \\(r = \\phi^{2\\theta/\\pi}\\) has the property that after each quarter turn (\\(\\theta\\) increases by \\(\\pi/2\\)), the radius is multiplied by \\(\\phi\\). Moreover, this spiral is self-similar: any sector is similar to the whole spiral.

Proof Sketch: For \\(r(\\theta) = \\phi^{2\\theta/\\pi} = e^{(2\\ln \\phi/\\pi)\\theta}\\), after \\(\\Delta \\theta = \\pi/2\\): \\[r(\\theta + \\pi/2) = \\phi^{2(\\theta + \\pi/2)/\\pi} = \\phi^{2\\theta/\\pi + 1} = \\phi \\cdot r(\\theta)\\] Self-similarity follows from the exponential form: scaling \\(r\\) by \\(\\phi^k\\) corresponds to rotating by \\(k\\pi/2\\), preserving the spiral's shape.

Notation Guide

  • \\(\\phi = \\frac{1+\\sqrt{5}}{2}\\): Golden ratio
  • \\(\\varphi = \\frac{1}{\\phi} = \\phi - 1\\): Reciprocal golden ratio
  • \\(F_n\\): \\(n\\)-th Fibonacci number
  • \\(\\Phi = \\begin{pmatrix} 1 & 1 \\ 1 & 0 \\end{pmatrix}\\): Fibonacci Q-matrix

📜 Math History: The Development of Abstract Algebra

The 19th century witnessed a profound transformation in mathematics: the shift from studying specific mathematical objects to studying the structures and relationships that govern them. This revolution, now called abstract algebra, emerged from the work of several mathematicians who sought to understand the common patterns underlying apparently disparate problems.

The story begins with Évariste Galois (1811-1832), whose work on the solvability of polynomial equations laid the foundation for group theory. Galois discovered that the symmetry group of a polynomial's roots determines whether it can be solved by radicals. His brilliant but tragically short life—he died in a duel at age 20—left behind manuscripts that would transform mathematics. The concept of a "group," implicit in Galois's work, would become one of the most fundamental structures in modern mathematics.

Meanwhile, Carl Friedrich Gauss was developing the theory of quadratic forms and modular arithmetic, introducing concepts that would later be recognized as part of ring theory. His "Disquisitiones Arithmeticae" (1801) established number theory as a rigorous discipline and introduced the notation and methods that would dominate the field for centuries.

The unification of these ideas came through the work of Richard Dedekind, who in the 1870s introduced the concepts of ideals and modules while studying algebraic number theory. Dedekind's abstract approach—defining structures by their properties rather than their representations—set the template for modern algebra. Emmy Noether, working in the early 20th century, completed this abstraction, showing that the properties of algebraic systems could be derived from simple axioms. Her famous dictum "Es steht alles schon bei Dedekind" ("It's all already in Dedekind") acknowledged her debt while demonstrating how much further abstraction could go.

By the mid-20th century, the structural approach had spread throughout mathematics. The Bourbaki group, founded in 1934, sought to rebuild mathematics from set-theoretic foundations, emphasizing structures like groups, rings, fields, and topological spaces. While their rigid approach eventually faced criticism, their influence on mathematical pedagogy and practice remains profound. IMO Shortlist Problem

Problem: Let \\(n\\) be a positive integer. A sequence of integers \\(a_0, a_1, ..., a_n\\) is called good if it satisfies the following conditions:

  • \\(a_0 = 0\\)
  • For \\(0 \\leq i \\leq n-1\\), we have \\(|a_{i+1} - a_i| = 1\\)
  • There do not exist indices \\(0 \\leq i < j \\leq n\\) such that \\(|a_j - a_i| = j - i\\)

Find the number of good sequences for a given \\(n\\).

💡 Hint

Consider the connection to Catalan numbers. The third condition prevents the sequence from having a "straight line" segment. Try small values of \\(n\\) and look for a pattern. For \\(n = 2\\), the valid sequences are \\((0, 1, 0)\\) and \\((0, -1, 0)\\).

✅ Solution

The answer is the \\(n\\)-th Catalan number \\(C_n = \\frac{1}{n+1}\\binom{2n}{n}\\).

Proof: We establish a bijection between good sequences and Dyck paths of length \\(2n\\).

First, observe that a sequence satisfying the first two conditions is essentially a walk on \\(\\mathbb{Z}\\) starting at 0, taking steps of \\(\\pm 1\\). The third condition says that no two points on the walk are connected by a straight line (i.e., we never have \\(a_j - a_i = \\pm(j-i)\\)).

This is equivalent to saying that the walk never touches the lines \\(y = x\\) or \\(y = -x\\) except at the origin. By reflecting appropriately, we can map this to a walk that stays above the x-axis, which corresponds to a Dyck path.

More precisely, define \\(b_i = a_i + i\\). Then \\(b_0 = 0\\) and \\(b_{i+1} - b_i \\in \\{0, 2\\}\\). The condition that \\(|a_j - a_i| \\neq j - i\\) becomes \\(b_j \\neq b_i\\) for \\(j > i\\), meaning all \\(b_i\\) are distinct. This creates a bijection with Dyck paths, counted by Catalan numbers.

🎨 Visualization of the Day: The Golden Spiral

The golden spiral emerges from nested golden rectangles, each containing a square that leaves another golden rectangle. As squares are successively removed, quarter-circle arcs connect the corners, forming a logarithmic spiral that appears throughout nature—from the arrangement of seeds in sunflowers to the shape of spiral galaxies.

The spiral's self-similarity reflects a fundamental mathematical property: it looks the same at any scale. This is because the golden ratio \\(\\phi\\) satisfies \\(\\phi^2 = \\phi + 1\\), creating a recursive structure that repeats infinitely. The connection to the Fibonacci sequence \\(F_n\\) comes from the identity \\(\\lim_{n\\to\\infty} \\frac{F_{n+1}}{F_n} = \\phi\\).

import numpy as np
import matplotlib.pyplot as plt
from matplotlib.patches import Rectangle, FancyBboxPatch

# Golden ratio
phi = (1 + np.sqrt(5)) / 2

fig, ax = plt.subplots(figsize=(10, 10))

# Generate nested golden rectangles
def draw_golden_spiral(ax, n_squares=8):
    x, y = 0, 0
    size = phi**n_squares
    
    colors = plt.cm.viridis(np.linspace(0, 1, n_squares))
    
    for i in range(n_squares):
        # Draw square
        square_size = size / phi**i
        rect = Rectangle((x, y), square_size, square_size,
                         linewidth=2, edgecolor='black',
                         facecolor=colors[i], alpha=0.3)
        ax.add_patch(rect)
        
        # Draw quarter circle arc
        theta = np.linspace(i * np.pi/2, (i+1) * np.pi/2, 50)
        r = square_size
        
        if i % 4 == 0:  # Bottom right
            cx, cy = x + square_size, y
            arc_x = cx - r * np.cos(theta - i*np.pi/2)
            arc_y = cy + r * np.sin(theta - i*np.pi/2)
            x, y = x + square_size/phi, y
        elif i % 4 == 1:  # Top right
            cx, cy = x, y + square_size
            arc_x = cx + r * np.cos(theta - i*np.pi/2)
            arc_y = cy - r * np.sin(theta - i*np.pi/2)
            x, y = x, y + square_size/phi
        elif i % 4 == 2:  # Top left
            cx, cy = x, y
            arc_x = cx + r * np.cos(theta - i*np.pi/2)
            arc_y = cy + r * np.sin(theta - i*np.pi/2)
            x, y = x - square_size/phi**2, y
        else:  # Bottom left
            cx, cy = x + square_size, y
            arc_x = cx - r * np.cos(theta - i*np.pi/2)
            arc_y = cy - r * np.sin(theta - i*np.pi/2)
            x, y = x, y - square_size/phi**2
        
        ax.plot(arc_x, arc_y, 'r-', linewidth=2)
    
    ax.set_xlim(-1, phi**n_squares + 1)
    ax.set_ylim(-1, phi**n_squares + 1)
    ax.set_aspect('equal')
    ax.axis('off')
    ax.set_title('Golden Spiral', fontsize=16, fontweight='bold')

draw_golden_spiral(ax, n_squares=6)
plt.tight_layout()
plt.savefig('golden_spiral.png', dpi=150, bbox_inches='tight',
            facecolor='white', edgecolor='none')
plt.show()
